import os, sys, time, gc
import napari, numpy as np
from qtpy.QtWidgets import QApplication, QRadioButton
v = napari.Viewer(show=True)
_, w = v.window.add_plugin_dock_widget("napari-labconstrictor", "LabConstrictor tools")
def pump(s=0.3):
    end=time.time()+s
    while time.time()<end: QApplication.instance().processEvents(); time.sleep(0.02)
def stray(): return [t for t in QApplication.topLevelWidgets() if t.isVisible() and isinstance(t, QRadioButton)]
pump(1.5); w.app_box.value="LabConstrictorPlayground"; pump()
def tool(prefix): w.tool_box.value=[l for l in w.tool_box.choices if l.startswith(prefix)][0]; pump(0.3)
def cycle(n):
    for k in range(n): v.add_image(np.zeros((16,16),np.uint8), name="p%d"%k); pump(0.2)
    v.layers.clear(); pump(0.3)
tool("Count"); print("Count only: start", len(stray()))
for n in (1,3,3,3,6): cycle(n); print(" cycle(%d) ->"%n, len(stray()), flush=True)
gc.collect(); pump(0.5); print("after gc.collect()", len(stray()))
tool("Feature"); cycle(3); print("Feature form after cycle(3):", len(stray()))
tool("Align"); cycle(3); print("Align form (no radios in it) after cycle(3):", len(stray()))
os._exit(0)
